--- 匹配算法核心逻辑（必须实现） ---
总分 = 硬性条件分（40分）+ 语义技能分（60分），满分100分。

1. 硬性条件分（40分）：
   a. 学历匹配（20分）：
      - 职位要求 "本科及以上" 或 "硕士及以上" → 求职者 "硕士" 或 "本科" 得 20分
      - 职位要求 "本科及以上" → 求职者 "专科" 得 10分
      - 职位要求 "本科及以上" → 求职者 "高中/中专" 得 0分
      - 职位要求 "硕士及以上" → 求职者 "硕士" 得 20分，"本科" 得 10分
      - 若职位无学历要求，默认所有人得 20分（视为门槛不限）

   b. 城市匹配（20分）：
      - 求职者期望城市 == 职位工作城市 → 20分
      - 同省不同市 → 10分（对应比赛额外加分项）
      - 不同省 → 0分
      - 若职位无城市要求，默认得 20分

2. 语义技能分（60分）：
   a. Jaccard系数（30分）：
      - 公式：(交集技能数 / 并集技能数) × 30
      - 若双方技能列表都为空，得 0分
      - 技能比较时不区分大小写

   b. 余弦相似度（30分）：
      - 简化版：使用词频向量（TF-IDF），计算夹角余弦值 × 30
      - 若技能列表为空，得 0分

### 学历匹配映射关系（直接写在代码里）

```java
private double calculateEducationScore(String resumeEdu, String jobEdu) {
    if (jobEdu == null || jobEdu.isEmpty()) return 20.0;
    
    Map<String, Integer> eduLevel = new HashMap<>();
    eduLevel.put("博士", 5);
    eduLevel.put("硕士", 4);
    eduLevel.put("本科", 3);
    eduLevel.put("专科", 2);
    eduLevel.put("高中", 1);
    eduLevel.put("中专", 1);
    
    int resumeLevel = eduLevel.getOrDefault(resumeEdu, 0);
    int jobLevel = eduLevel.getOrDefault(jobEdu, 0);
    
    if (resumeLevel >= jobLevel) return 20.0;
    if (resumeLevel == jobLevel - 1) return 10.0;
    return 0.0;
}

private double calculateCityScore(String resumeCity, String jobCity) {
    if (jobCity == null || jobCity.isEmpty()) return 20.0;
    if (resumeCity == null || resumeCity.isEmpty()) return 0.0;
    if (resumeCity.equals(jobCity)) return 20.0;
    // 同省判断（简单版：取城市名的前2个字符，如"成都"和"绵阳"都属于"四川"）
    // 实际项目中可用省份映射表，这里简化处理
    if (resumeCity.substring(0, 2).equals(jobCity.substring(0, 2))) return 10.0;
    return 0.0;
}

private double calculateJaccardScore(List<String> resumeSkills, List<String> jobSkills) {
    if (resumeSkills == null || jobSkills == null) return 0.0;
    if (resumeSkills.isEmpty() && jobSkills.isEmpty()) return 0.0;
    Set<String> set1 = new HashSet<>(resumeSkills);
    Set<String> set2 = new HashSet<>(jobSkills);
    Set<String> intersection = new HashSet<>(set1);
    intersection.retainAll(set2);
    Set<String> union = new HashSet<>(set1);
    union.addAll(set2);
    return (intersection.size() / (double) union.size()) * 30.0;
}

private double calculateCosineScore(List<String> resumeSkills, List<String> jobSkills) {
    if (resumeSkills == null || jobSkills == null) return 0.0;
    if (resumeSkills.isEmpty() || jobSkills.isEmpty()) return 0.0;
    // 构建词频向量（简单计数）
    Map<String, Integer> resumeFreq = new HashMap<>();
    Map<String, Integer> jobFreq = new HashMap<>();
    for (String skill : resumeSkills) {
        resumeFreq.put(skill.toLowerCase(), resumeFreq.getOrDefault(skill.toLowerCase(), 0) + 1);
    }
    for (String skill : jobSkills) {
        jobFreq.put(skill.toLowerCase(), jobFreq.getOrDefault(skill.toLowerCase(), 0) + 1);
    }
    Set<String> allSkills = new HashSet<>(resumeFreq.keySet());
    allSkills.addAll(jobFreq.keySet());
    double dotProduct = 0.0;
    double resumeNorm = 0.0;
    double jobNorm = 0.0;
    for (String skill : allSkills) {
        int rf = resumeFreq.getOrDefault(skill, 0);
        int jf = jobFreq.getOrDefault(skill, 0);
        dotProduct += rf * jf;
        resumeNorm += rf * rf;
        jobNorm += jf * jf;
    }
    if (resumeNorm == 0 || jobNorm == 0) return 0.0;
    return (dotProduct / (Math.sqrt(resumeNorm) * Math.sqrt(jobNorm))) * 30.0;
}
